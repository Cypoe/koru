### 030_010_array_literal_simple

```koru
// TEST: Simple array literal syntax in flow arguments
//
// Koru should support clean array literal syntax: [a, b, c]
// The compiler infers the element type from the parameter type.
//
// Expected: Compiles and runs, passing array to the event

const std = @import("std");

tor sum { numbers: []const i32 } -> i32

proc sum|zig {
    var total: i32 = 0;
    for (numbers) |n| {
        total += n;
    }
    return total;
}

tor check { expected: i32, actual: i32 }

proc check|zig {
    if (expected != actual) {
        std.debug.print("FAIL: expected {}, got {}\n", .{ expected, actual });
        std.process.exit(1);
    }
    std.debug.print("PASS: {} == {}\n", .{ expected, actual });
}

// Use array literal syntax
sum(numbers: [1, 2, 3, 4]): r |> check(expected: 10, actual: r)
```

## CONTROL FLOW

### 240_subflow_defines_semantics

```koru
// Test 240: subflows define branch semantics
//
// Branch names in Koru are inert identifiers. Their meaning — what
// "return", "break", "continue", "ok", "north", or any other name
// signifies — is defined by the SUBFLOWS that resolve them, not by
// the compiler. This is the structural advantage over languages that
// bake fixed semantics for certain names into their grammar.
//
// This test deliberately uses names that ARE keywords in other
// languages (return, break, continue), to verify that the parser
// carries no value judgment about them. If a "common mistakes from
// other languages" trap is re-introduced at parser.zig parseStep,
// this test fails. That is the point.

import std/io

// Lower-level event: arbitrary outcome names
pub tor step {}
| return
| break
| continue

step => continue

// Outer event: its own outcome vocabulary
pub tor run {}
| stopped
| iterated

// Subflow: `run` is satisfied by `step`, with `step`'s outcomes
// mapped to `run`'s. The meaning of `return`, `break`, `continue`
// lives HERE — in user code, not in the compiler.
run = step()
| return => stopped
| break => stopped
| continue => iterated

std/io:print.ln("Testing subflow-defined semantics:")
run()
| stopped |> std/io:print.ln("  stopped")
| iterated |> std/io:print.ln("  iterated")
```

**Output:**

```
Testing subflow-defined semantics:
  iterated
```

### 208_simple_server_flow

```koru
// ============================================================================
// VERIFIED REGRESSION TEST - DO NOT MODIFY WITHOUT DISCUSSION
// ============================================================================
// Test 208: Simple server-style flow with nested labels
// A simplified synthetic example inspired by tcp-echo for flow visualization
// ============================================================================

const std = @import("std");

// Server events
tor listen { port: u16 }
| ready u32
| failed string

proc listen|zig {
    std.debug.print("Listening on port {}\n", .{port});
    if (port == 0) {
        return .{ .failed = "Invalid port" };
    }
    return .{ .ready = port };
}

tor accept { server: u32 }
| connected { conn: u32, server: u32 }
| failed u32

proc accept|zig {
    std.debug.print("Accepting connection\n", .{});
    if (server == 9999) {
        return .{ .failed = server };
    }
    return .{ .connected = .{ .conn = server + 1, .server = server } };
}

tor process { conn: u32 }
| done u32
| retry u32
| error { conn: u32, msg: string }

proc process|zig {
    std.debug.print("Processing connection {}\n", .{conn});
    if (conn % 3 == 0) {
        return .{ .error = .{ .conn = conn, .msg = "Processing error" } };
    }
    if (conn % 2 == 0) {
        return .{ .done = conn };
    }
    return .{ .retry = conn };
}

tor close { conn: u32 }

proc close|zig {
    std.debug.print("Closed connection {}\n", .{conn});
}

tor log-error { conn: u32, msg: string } -> u32

proc log-error|zig {
    std.debug.print("Error on connection {}: {s}\n", .{conn, msg});
    return conn;
}

tor cleanup { conn: u32 }

proc cleanup|zig {
    std.debug.print("Cleaning up connection {}\n", .{conn});
}

// Main flow with nested labels
listen(port: 8080)
| ready s |> #accept_loop accept(server: s)
    | connected c |> #process_loop process(c.conn)
        | done d |> close(conn: d) |> @accept_loop(c.server)
        | retry r |> @process_loop(conn: r)
        | error e |> log-error(e.conn, e.msg): l |> cleanup(conn: l) |> @accept_loop(c.server)
    | failed f |> @accept_loop(server: f)
| failed _ |> _
```

### 320_137_cond_constructs_named_branches

```koru
// Pins: cond arms construct named event branches with `=>` (same glyph as
// `if | then => more` in 020_028 / 240). Value-produce `cond` + `->` is
// 320_133; this pin is the branch-constructor twin for app routers
// (page-route → | catalog | progress | …).
import std/io

tor page-route { page: i64 }
| catalog
| progress
| other

page-route = cond(page)
| c p when p == 0 => catalog
| c p when p == 1 => progress
| c _ => other

page-route(page: 0)
| catalog |> std/io:print.ln("catalog")
| progress |> std/io:print.ln("progress")
| other |> std/io:print.ln("other")

page-route(page: 1)
| catalog |> std/io:print.ln("catalog")
| progress |> std/io:print.ln("progress")
| other |> std/io:print.ln("other")

page-route(page: 9)
| catalog |> std/io:print.ln("catalog")
| progress |> std/io:print.ln("progress")
| other |> std/io:print.ln("other")
```

**Output:**

```
catalog
progress
other
```

# RUNTIME FEATURES

### 400_130_effect_subflow_fires_void_arm

```koru
// SUBFLOW-IMPLEMENTED EFFECTS, minimal case (pure .k, no proc) — the subflow
// sibling of 400_070. `ping` declares one void-ish effect arm and NO terminal
// branches; its impl is a SUBFLOW that FIRES the arm by CALLING it, exactly as
// a `proc` body calls `pong(msg)` today. Ruled 2026-07-02: firing introduces
// NO new grammar — inside the impl of the declaring event, the event's own
// arms are callable, and everything downstream reuses the established subflow
// forms. Here that's the simplest one: the impl body is a plain call head
// (`ping = pong(x)`, same shape as `run = step()` in 240).
//
// The prize over the proc version: the firing is now a KORU act the flow
// checker can see, instead of an opaque host call trusted "unsafe"-style.
import std/io

pub tor ping { x: i64 }
! pong i64

ping = pong(x)

ping(x: 41)
! pong reply |> std/io:print.ln("{{ reply + 1:d }}")
```

**Output:**

```
42
```
